<script>
  (function() {
    var posts = [{% for post in site.posts %}{ "url": {{ post.url | relative_url | jsonify }}, "title": {{ post.title | jsonify }} }{% unless forloop.last %}, {% endunless %}{% endfor %}];

    /* GitHub Pages URLs are case-sensitive and the older permalinks mix case, "_"/"-" and the
       trailing slash, so compare loosely: the whole path first, then just the last segment. */
    function key(path) {
      try { path = decodeURIComponent(path); } catch (e) {}
      return path.toLowerCase().replace(/\.html$/, '').replace(/[_\s]+/g, '-').replace(/\/+$/, '');
    }
    function last(path) { return key(path).split('/').pop(); }

    var here = key(location.pathname);
    var match = posts.filter(function(p) { return key(p.url) === here; })[0] ||
                posts.filter(function(p) { return last(p.url) === last(location.pathname); })[0];
    if (match && here) {
      var box = document.getElementById('not-found-suggestion');
      var link = box.querySelector('a');
      link.href = match.url;
      link.textContent = match.title;
      box.hidden = false;
    }

    document.getElementById('not-found-search').addEventListener('click', function() {
      var toggle = document.querySelector('.search__toggle');
      if (toggle) toggle.click();
    });
  })();
</script>
